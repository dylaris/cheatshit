<<<
# 类定义
class Foo:
    # 类属性
    class_attr = 10

    # 构造函数
    def __init__(self, param1, param2):
        self.public = param1      # 公有属性
        self._protected = param2  # 保护属性
        self.__private = param2   # 私有属性（名称修饰）

    # property（getter/setter）
    @property
    def name(self):
        return self._name

    @name.setter
    def name(self, value):
        self._name = value

    # 实例方法
    def method(self, param1, param2):
        return param1 + param2

    # 类方法
    @classmethod
    def class_method(cls, param):
        return param

    # 静态方法
    @staticmethod
    def static_method(param):
        return param

    # 私有方法
    def __private_method(self):
        return "private"

# 对象字面量（使用字典或 SimpleNamespace）
from types import SimpleNamespace
obj = SimpleNamespace(prop="value")

# 实例化
a = Foo(param1, param2)
>>>

##################################################
# 函数
##################################################
<<<定义
def add(a, b):
    return a + b

# 默认参数
def greet(name="world"):
    return f"hello {name}"

# 关键字参数
def func(a, b, c=0, *args, **kwargs):
    pass

# 返回值类型注解
def add(a: int, b: int) -> int:
    return a + b

# 匿名函数（lambda）
add = lambda a, b: a + b

# 装饰器
def logger(func):
    def wrapper(*args, **kwargs):
        print(f"calling {func.__name__}")
        return func(*args, **kwargs)
    return wrapper

@logger
def say_hello():
    print("hello")
>>>

<<<调用
add(2, 3)
add(a=2, b=3)
func(1, 2, 3, 4, 5, x=10, y=20)
>>>

<<<闭包
def counter():
    count = 0
    def inner():
        nonlocal count
        count += 1
        return count
    return inner
>>>

##################################################
# 错误处理
##################################################
<<<try-except
try:
    # 可能出错的代码
    raise ValueError("something wrong")
except ValueError as e:
    print(e)
except (TypeError, ZeroDivisionError):
    pass
except Exception as e:
    print(f"unexpected: {e}")
else:
    # 没有异常时执行
    pass
finally:
    # 总是执行
    pass
>>>

<<<自定义异常
