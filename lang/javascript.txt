# 复合类型
##################################################
class                             # 类
Object                            # 对象

<<<
// 类定义
class Foo {
  // 静态属性
  static staticProp = 10;

  // 私有字段（必须以#开头）
  #privateField;

  // 构造函数
  constructor(param1, param2) {
    this.publicProp = param1;    // 公有属性
    this.#privateField = param2; // 私有属性
  }

  // getter
  get name() {
    return this._name;
  }

  // setter
  set name(value) {
    this._name = value;
  }

  // 静态getter
  static get version() {
    return "1.0.0";
  }

  // 实例方法
  method(param1, param2) {
    return param1 + param2;
  }

  // 静态方法
  static staticMethod(param) {
    return param;
  }

  // 私有方法
  #privateMethod() {
    return "private";
  }
}

// 对象字面量
const obj = {
  prop: "value",
  method() {
    return this.prop;
  },
  get computed() {
    return this.prop;
  }
};

let a = new Foo(param1, param2);
>>>

##################################################
# 数组/列表
##################################################
<<<创建
const arr = [];
const arr = [1, 2, 3, 4];
const arr = new Array(5);        // 长度为5的空数组
const arr = Array.from("hello"); // ['h','e','l','l','o']
const arr = [...iterable];       // 展开运算符
>>>

<<<增
arr.push(10);                    // 末尾添加
arr.unshift("head");             // 开头添加
arr.splice(2, 0, "insert");      // 指定位置插入
arr = arr.concat(["a", "b"]);    // 合并数组
>>>

<<<删
arr.pop();                       // 删除末尾
arr.shift();                     // 删除开头
arr.splice(2, 1);                // 删除索引2开始的1个元素
delete arr[2];                   // 删除但保留空位
arr.length = 0;                  // 清空数组
arr = [];                        // 清空数组
>>>

<<<查
let length = arr.length;
let element = arr[2];
let index = arr.indexOf("value");
let lastIndex = arr.lastIndexOf("value");
let found = arr.find(x => x > 10);
let includes = arr.includes("value");
let slice = arr.slice(1, -1);
>>>

<<<改
arr[0] = "first";
arr.fill(0, 2, 4);               // 填充索引2-4为0
arr.reverse();                   // 反转
arr.sort();                      // 排序
arr.sort((a,b) => a - b);        // 数字升序
>>>

<<<遍历
for (let i of arr) {
  // ...
}

arr.forEach((item, index) => {
  // ...
});

for (let i = 0; i < arr.length; i++) {
  // ...
}
>>>

<<<高阶函数
let doubled = arr.map(x => x * 2);
let evens = arr.filter(x => x % 2 === 0);
let sum = arr.reduce((acc, x) => acc + x, 0);
let allPositive = arr.every(x => x > 0);
let somePositive = arr.some(x => x > 0);
>>>

##################################################
# 函数
##################################################
<<<函数声明
function add(a, b) {
  return a + b;
}

// 函数表达式
const add = function(a, b) {
  return a + b;
};

// 箭头函数
const add = (a, b) => a + b;
const greet = name => `hello ${name}`;
const getObj = () => ({ key: "value" });

// 默认参数
function greet(name = "world") {
  return `hello ${name}`;
}

// 剩余参数
function sum(...numbers) {
  return numbers.reduce((a, b) => a + b, 0);
}

// 立即执行函数（IIFE）
(function() {
  // ...
})();

(() => {
  // ...
})();
>>>

<<<调用
add(2, 3);
>>>

<<<高阶函数
function operate(a, b, fn) {
  return fn(a, b);
}

// 闭包
function counter() {
  let count = 0;
  return () => ++count;
}
>>>

##################################################
# 错误处理
##################################################
<<<try-catch
try {
  // 可能出错的代码
  throw new Error("something wrong");
} catch (error) {
  console.log(error.message);
  console.log(error.stack);
} finally {
  // 总是执行
}
>>>

<<<自定义错误
class CustomError extends Error {
  constructor(message) {
    super(message);
    this.name = "CustomError";
  }
}

throw new CustomError("custom error");
>>>

<<<Promise错误处理
promise
  .then(result => {})
  .catch(error => {})
  .finally(() => {});

async function example() {
  try {
    const result = await promise;
  } catch (error) {
    // 处理错误
  }
}
>>>
