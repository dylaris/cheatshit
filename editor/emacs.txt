##################################################
# 光标移动
##################################################
C-f                   # 光标前进一个字符
C-b                   # 光标后退一个字符
C-n                   # 光标下移一行
C-p                   # 光标上移一行
M-f                   # 跳到下一个单词末尾
M-b                   # 跳到上一个单词开头
C-a                   # 跳到行首
C-e                   # 跳到行尾

##################################################
# 编辑操作
##################################################
C-d                   # 删除光标后一个字符
M-d                   # 删除光标后整个单词
DEL                   # 删除光标前一个字符
M-DEL                 # 删除光标前整个单词
C-k                   # 从光标处删除到行尾
C-u                   # 从光标处删除到行首
C-y                   # 粘贴（召回删除内容）
C-_                   # 撤销
C-x u                 # 撤销

##################################################
查找替换
##################################################
C-s                   # 正向增量搜索
C-r                   # 反向增量搜索
