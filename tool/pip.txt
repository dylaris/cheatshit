##################################################
# 帮助与版本
##################################################
pip --version               # 查看版本
pip -V                      # 同上（简写）
pip --help                  # 查看帮助
pip help install            # 查看特定命令帮助


##################################################
# 安装包
##################################################
pip install <package>       # 安装最新版
                            # ==1.2.3      指定版本
                            # -U           升级
                            # --user       仅当前用户
                            # --no-deps    不装依赖
                            # --no-cache-dir 不用缓存
pip install -r requirements.txt   # 批量安装


##################################################
# 卸载包
##################################################
pip uninstall <package>     # 卸载
                            # -y, --yes    自动确认


##################################################
# 升级包
##################################################
pip install -U <package>    # 升级到最新版
pip install -U pip          # 升级 pip 自身


##################################################
# 列出已安装包
##################################################
pip list                    # 列出所有包
                            # --outdated   列出可升级的包
                            # -o           同上（简写）
                            # --uptodate   列出最新的包
                            # --format=json  JSON格式


##################################################
# 查看包信息
##################################################
pip show <package>          # 显示详细信息
                            # -f, --files  显示包内文件


##################################################
# requirements.txt
##################################################
pip freeze > requirements.txt       # 导出当前环境所有包
pip install -r requirements.txt     # 从文件安装
pip uninstall -r requirements.txt -y # 卸载文件中的所有包


##################################################
# 换源（镜像加速）
##################################################
pip install -i <镜像源> <package>   # 临时使用
# 常用镜像源:
#   清华: https://pypi.tuna.tsinghua.edu.cn/simple
#   阿里: https://mirrors.aliyun.com/pypi/simple
#   豆瓣: https://pypi.douban.com/simple


##################################################
# 缓存管理
##################################################
pip cache info              # 查看缓存信息
pip cache purge             # 清除所有缓存


##################################################
# 检查依赖
##################################################
pip check                   # 检查已安装包的依赖冲突


##################################################
# 查看与修改配置
##################################################
pip config list             # 查看当前所有配置
pip config list --user      # 查看用户级配置
pip config list --global    # 查看全局配置
pip config list --site      # 查看环境级配置

pip config get <key>        # 查看指定配置项
                            # 示例: pip config get global.index-url

pip config set <key> <val>  # 设置配置项
                            # 示例: pip config set global.index-url https://pypi.tuna.tsinghua.edu.cn/simple
                            # 示例: pip config set install.trusted-host pypi.tuna.tsinghua.edu.cn

pip config unset <key>      # 删除配置项（恢复默认）
                            # 示例: pip config unset global.index-url

pip config edit             # 编辑配置文件（打开编辑器）
                            # --user      编辑用户级配置
                            # --global    编辑全局配置
